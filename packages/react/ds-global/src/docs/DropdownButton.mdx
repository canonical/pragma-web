# DropdownButton

A Button with a persistent chevron that opens a [ContextualMenu](?path=/docs/documentation-contextualmenu--docs).

```tsx
import { DropdownButton } from "@canonical/react-ds-global";

<DropdownButton
  importance="secondary"
  items={[
    { key: "rename", label: "Rename" },
    { type: "separator", key: "s" },
    { url: "/docs", label: "Docs" },
  ]}
  onSelect={(item) => run(item.key)}
>
  Actions
</DropdownButton>;
```
